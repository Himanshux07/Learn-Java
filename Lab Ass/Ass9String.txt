def naive_string_matcher(text:str,pattern:str):
    n=len(text)
    m=len(pattern)

    found_indices=[]

    if(n==0 or m==0 or m>n):
        return found_indices
    
    for i in range(n-m+1):
        is_match=True
        for j in range(m):
            if text[i+j]!=pattern[j]:
                is_match=False
                break
        if is_match:
            found_indices.append(i)

    return found_indices

str="ababcabcabababd"
str2="abc"

print(naive_string_matcher(str,str2))
