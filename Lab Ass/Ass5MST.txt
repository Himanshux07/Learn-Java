def find(parent, i):
    
    if parent[i] != i:
        parent[i] = find(parent, parent[i])  # Path compression
    return parent[i]

def union(parent, rank, x, y):
    """Union two sets by rank"""
    xroot = find(parent, x)
    yroot = find(parent, y)

    if rank[xroot] < rank[yroot]:
        parent[xroot] = yroot
    elif rank[xroot] > rank[yroot]:
        parent[yroot] = xroot
    else:
        parent[yroot] = xroot
        rank[xroot] += 1

def kruskal(n, edges):
    edges.sort(key=lambda x: x[2])  # Sort edges by weight
    parent = [i for i in range(n)]
    rank = [0] * n
    mst = []
    total_weight = 0

    for u, v, w in edges:
        root_u = find(parent, u)
        root_v = find(parent, v)

        if root_u != root_v:  # If it doesn’t form a cycle
            mst.append((u, v, w))
            total_weight += w
            union(parent, rank, root_u, root_v)

    print("\n🔹 Minimum Spanning Tree using Kruskal’s Algorithm:")
    for u, v, w in mst:
        print(f"{u} -- {v} == {w}")
    print("Total Weight of MST:", total_weight)

def prim(n, edges):
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((w, v))
        graph[v].append((w, u))

    visited = [False] * n
    pq = [(0, 0)]  # (weight, vertex)
    total_weight = 0
    mst = []

    while pq:
        w, u = heapq.heappop(pq)
        if visited[u]:
            continue
        visited[u] = True
        total_weight += w
        for edge in graph[u]:
            if not visited[edge[1]]:
                heapq.heappush(pq, edge)
        if w != 0:
            mst.append((u, w))

    print("\n🔹 Minimum Spanning Tree using Prim’s Algorithm:")
    for u, w in mst:
        print(f"Vertex {u} connected with edge weight {w}")
    print("Total Weight of MST:", total_weight)
